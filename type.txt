typedef struct s_player
{
	double pos_x;	   // Position X du joueur dans la map (en cases, ex: 2.5)
	double pos_y;	   // Position Y du joueur dans la map (en cases, ex: 3.7)
	double dir_x;	   // Composante X du vecteur direction (où le joueur regarde)
	double dir_y;	   // Composante Y du vecteur direction (où le joueur regarde)
	double plane_x;	   // Composante X du plan caméra (perpendiculaire à dir, définit le FOV)
	double plane_y;	   // Composante Y du plan caméra (perpendiculaire à dir, définit le FOV)
	double move_speed; // Vitesse de déplacement (en cases/frame, ex: 0.05)
	double rot_speed;  // Vitesse de rotation (en radians/frame, ex: 0.03)
} t_player;

typedef struct s_ray
{
	double camera_x;	   // Position X du rayon sur le plan caméra, de -1 (gauche) à 1 (droite)
	double ray_dir_x;	   // Composante X de la direction du rayon courant
	double ray_dir_y;	   // Composante Y de la direction du rayon courant
	int map_x;			   // Case X actuelle du rayon dans la grille de la map
	int map_y;			   // Case Y actuelle du rayon dans la grille de la map
	double side_dist_x;	   // Distance jusqu'au prochain bord vertical de case (axe X)
	double side_dist_y;	   // Distance jusqu'au prochain bord horizontal de case (axe Y)
	double delta_dist_x;   // Distance entre deux bords verticaux consécutifs (pas en X)
	double delta_dist_y;   // Distance entre deux bords horizontaux consécutifs (pas en Y)
	double perp_wall_dist; // Distance perpendiculaire au plan caméra (évite le fish-eye)
	int step_x;			   // Direction du pas DDA en X : +1 ou -1
	int step_y;			   // Direction du pas DDA en Y : +1 ou -1
	int hit;			   // 1 si le rayon a touché un mur, 0 sinon
	int side;			   // 0 = mur NS touché (face E/W), 1 = mur EW touché (face N/S)
	int line_height;	   // Hauteur en pixels de la colonne de mur à afficher
	int draw_start;		   // Pixel Y où commencer à dessiner la colonne
	int draw_end;		   // Pixel Y où arrêter de dessiner la colonne
	int wall_dir;		   // Direction du mur touché : N/S/E/W → index de texture (0-3)
} t_ray;

typedef struct s_texture
{
	void *img;	  // Pointeur sur l'image MLX (retourné par mlx_new_image ou mlx_xpm_file_to_image)
	char *addr;	  // Adresse mémoire brute des pixels (retournée par mlx_get_data_addr)
	int width;	  // Largeur de l'image en pixels
	int height;	  // Hauteur de l'image en pixels
	int bpp;	  // Bits par pixel (bits_per_pixel, généralement 32)
	int line_len; // Taille d'une ligne en octets (line_length retourné par mlx_get_data_addr)
	int endian;	  // Ordre des octets de couleur (0 = little-endian, 1 = big-endian)
} t_texture;

typedef struct s_game
{
	void *mlx;			   // Pointeur de connexion MLX (retourné par mlx_init)
	void *win;			   // Pointeur sur la fenêtre (retourné par mlx_new_window)
	t_texture screen;	   // Image buffer principal : tout le rendu se fait ici avant mlx_put_image_to_window
	t_texture textures[4]; // Les 4 textures de murs : [0]=NO, [1]=SO, [2]=EA, [3]=WE
	t_player player;	   // Struct joueur : position, direction, vitesse
	t_ray ray;			   // Struct rayon courant : recalculée à chaque colonne de pixel
	char **map;			   // La grille de la map (tableau 2D de chars : '0', '1', 'N', etc.)
	int map_width;		   // Largeur maximale de la map en cases
	int map_height;		   // Hauteur de la map en cases (nombre de lignes)
	int floor_color;	   // Couleur du sol encodée en int 0xRRGGBB
	int ceil_color;		   // Couleur du plafond encodée en int 0xRRGGBB
	int **doors;		   // Tableau 2D des états des portes : 0=fermée, 1=ouverte (bonus)
	int mouse_enabled;	   // 1 si la rotation souris est active, 0 sinon (bonus)
	int mouse_x_prev;	   // Dernière position X de la souris pour calculer le delta de rotation (bonus)
} t_game;
